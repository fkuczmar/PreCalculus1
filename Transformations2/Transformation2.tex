\section{Applications}

\begin{question} \label{Qf330zxal}
The function
\[
     A = f(t) \, , \, 0\leq t \leq 4,
\]
expresses the altitude of a balloon (in meters) in terms of the number of hours past noon.

The range of the function $f$ is
\[
 {\cal R}_f = \{ A \, | \, 500 \leq A \leq 2000  \}.
\]

\begin{enumerate}

\item Find expressions for each of the following functions. Start by stating the domain and range of each.

\begin{enumerate}
\item The function 
\[
     H = g(m) ,
\]
that expresses the altitude of the balloon, measured in kilometers, in terms of the number of minutes past noon. 

\item The function $A = k(s)$ that expresses the balloon's altitude (in meters) in terms of the number of hours past 3pm. 

\item The function $A =p(s)$ that expresses the balloon's altitude (in meters) in terms of the number of minutes past 3pm. 

\item The funtion $B = q(t)$ that expresses the signed height (meters) of the balloon relative to the top of the Space Needle (700 feet above sea level) in terms of the number of hours past noon. Take the height to be negative when the balloon is below the top of the Space Needle.

\item The funtion $C = r(t)$ that expresses the signed height (km) of the balloon relative to the top of the Space Needle in terms of the number of minutes past 3pm. Take the height to be negative when the balloon is below the top of the Space Needle.

\end{enumerate}

\item Suppose the point with coordinates $(a,b)$ lies on the graph of the function $A=f(t)$. Find the coordinates of the corresponding point on the graphs of the five functions in part (a).

\end{enumerate}
\end{question}

The function
\[
    G = f(v) \, , \, 20\leq v \leq 100,
\]
expresses the gas mileage of a car (measured in km/Liter) in terms of its speed (measured in km/hour). 

The range of the function $f$ is
\[
  {\cal R}_f = \{ G \, | \, 3 \leq G \leq  18 \}
\]



\begin{enumerate}
\item Find expressions for each of the following functions. Start by stating the domain of each. Assume there are exactly $1.6$ km in one mile and exactly $4$ liters in one gallon. 

\begin{enumerate}
\item The function $K = h(v)$ that expresses the car's gas mileage (in miles/gallon) in terms of its speed (in km/hour).

\item The function $K=k(s)$ that expresses the car's gas mileage (in miles/gallon) in terms of its speed (in miles/hour).
\end{enumerate}

\item Suppose the point with coordinates $(a,b)$ lies on the graph of the function $A=f(t)$. Find the coordinates of the corresponding points on the graphs of the two functions in part (a).


\end{enumerate}
\end{question}


\section{Circles}

\begin{question} \label{Qpdf3r000}
The point with coordinates $(a,b)$ lies on the curve with equation
\[
  x^2 + y^2 = 25.
\]

\begin{enumerate}
\item Find the coordinates of the corresponding point on each of the following curves.

\begin{enumerate}
\item 
\[
   (x-3)^2 + (y-2)^2 = 25.
\]

\item 
\[
   (x+3)^2 + (y+2)^2 = 25.
\]

\item 
\[
   \left( \frac{x}{3} \right)^2 + \left( \frac{y}{3} \right)^2 = 25.
\]

\item 
\[
   \left( \frac{x}{3} \right)^2 + \left( \frac{y}{2} \right)^2 = 25.
\]

\item 
\[
   \left( \frac{x-6}{3} \right)^2 + \left( \frac{y+7}{3} \right)^2 = 25.
\]

\item 
\[
   \left( \frac{x-6}{3} \right)^2 + \left( \frac{y+7}{2} \right)^2 = 25.
\]

\end{enumerate}

\item Find the domain and range of the relation $x^2 + y^2 = 25$ and the domains and ranges of the six relations in part (a).

\item Describe the transformations that take the curve $x^2 + y^2 = 25$ to each of the six curves in part (a).

\item Graph each of the curves in part (a). Label the coordinates of the point corresponding to the point with coordinates $(3,4)$ on the curve $x^2 + y^2 = 25$.
\end{enumerate}

\end{question}


\section{Parabolas}

\begin{question} \label{Qdrer3f39}

The point with coordinates $(a,b)$ lies on the parabola
\[
   h = f(t) = t^2 \, , \, -3 \leq t \leq 4.
\]

\begin{enumerate}
\item Find the coordinates of the correponding point on each of the following curves.

\begin{enumerate}
     \item $h = -5f(t)$

     \item $h=80 - 5f(t)$ 

      \item $h=80 - 5f(t-3)$

\item Find the domain and range of each of the functions in part (a).

\item Graph each function in part (a). Label the coordinates of the points corresponding to $(-3,9)$, $(0,0)$, and $(4,16)$ on each.

\item  Describe the transformations that take the graph of the function $h=f(t)$ to the graphs of each of the functions in part (a).

\end{enumerate}
\end{enumerate}
\end{question}
