\section{Quiz 1B}

\begin{question} \label{Qpfo3r}
Find an equation of the smallest circle through the points with coordinates $(3,-1)$ and $(-7,-9)$. 
\begin{itemize}

\item Include an accurate picture on a coordinate system. Label the axes with the appropriate variable names. Put scales on the axes.


\item Include a complete explanation in full sentences of each step in your solution. Do not save the explanation to the end.

\item Explain the \emph{logic} behind your equation. Use the Pythagorean theorem in your explanation, not the distance formula.

Note: Explaining the logic is not the same as explaining how to use a formula. It requires a lot more.

\end{itemize}

\begin{explanation}
I'll leave the pictures to you, but the key points are as follows:

\begin{enumerate}
\item Since the longest chord of a circle is a diameter, the smallest circle through the points $A(3,-1)$ and $B(-7,-9)$ is the circle that has segment $\overline{AB}$ as a diameter.

\item The center of that circle is the midpoint $M$ of $\overline{AB}$. To get the coordinates of $M$ we average the coordinates of $A$ and $B$. The midpoint therefore has coordinates
\[
  (x,y) = \left( \frac{1}{2}\left(3+ -7  \right) , \frac{1}{2}\left(-1+ -9  \right) \right) = (-2.,-5).
\]

\item The radius of the smallest circle is the distance between $M$ and either of the points $A$ or $B$. We'll find the distance between $M$ and $A$ by using the Pythagorean theorem. To do this we draw a horizontal line through $M$ and a vertical line through $A$. These lines intersect in the point $C$ with coordinates $(3,-5)$. Then the legs of right triangle $ACM$ have lengths
\[
    a = \text{dist}(M,C) = |3 - (-2)| = 5
\]
and
\[
   m = \text{dist}(A,C) = |-5 - (-1) | = 4 .
\]
So by the Pythagorean theorem the hypotenuse $\overline{AM}$ has length
\[
    c =  \sqrt{a^2 + m^2} = \sqrt{25 + 16} = \sqrt{41}.
\]

\item We can now write an equation of the circle centered at $M$ with radius $\sqrt{41}$. The key idea behind the equation is this.
 
\pskip

A point $P$ lies on the circle if and only if the distance from $P$ to the center $M$ equals the radius $\sqrt{41}$. 

\pskip

To turn this desription of the circle into an equation, we let $P$ have coordinates $(x,y)$. Then by drawing a right triangle and using the Pythagorean theorem as we did in part (c), the distance from $P(x,y)$ to the center $M(-2,-5)$ is
\[
      \sqrt{(x+2)^2 + (y+5)^2}.
\]

So a point $P$ with coordinates $(x,y)$ lies on the circle of radius $\sqrt{41}$ centered at $M(-2,-5)$ if and only if
\[
  \sqrt{(x+2)^2 + (y+5)^2} = \sqrt{41}.
\]

This is an equation of the smallest circle through the points $A(3,-1)$ and $B(-7,-9)$.

\end{enumerate}

\emph{Comment:} The most common error was failing to include an explanation of the \emph{logic} behind the equation of the circle as in part (d).
\end{explanation}
\end{question}

\section{Quiz 2A}
\begin{question} \label{Qpv3ffg}
 Let
\[
     f(x) = \frac{6}{7} \left( x - 5 \right)^2 - 4 .
\]

\begin{enumerate}
\item Desribe the action of $f$ on its input using complete sentences.

\item Use the methods of this class so far to solve the equation
\[
     f(x) = f(-2) .
\]
\end{enumerate}

\begin{explanation}
\item The question is asking us to desribe what $f$ does to its input. 
\begin{itemize}
\item The function $f$ first subtracts $5$ from its input.

\item The function then squares the result.

\item It then multiplies the previous result by $6/7$.

\item And finally $f$ subtracts $4$.
\end{itemize}

\item To solve the equation $f(x) = f(-2)$ means to solve the equation
\[
       \frac{6}{7} \left( x - 5 \right)^2 - 4 = \frac{6}{7} \left( 2 - 5 \right)^2 - 4 .
\]
Adding $4$ to both sides gives the equivalent equation
\[
  \frac{6}{7} \left( x - 5 \right)^2 = \frac{6}{7} \left( 2 - 5 \right)^2 .
\]
And this equation is true (multiply both sides by $6/7$) if and only if
\[
   \left( x - 5 \right)^2 = \left( 2 - 5 \right)^2 ,
\] 
or if and only if
\[
  (x-5)^2 = 9 .
\]
So either 
\[
   x-5 = 3 \hskip 0.2 in \text{or} \hskip 0.2 in x-5 = -3,
\]
and
\[
      x = 8 \hskip 0.2 in \text{or} \hskip 0.2 in x= -2.
\]
So the solution set to the equation $f(x)=f(-2)$ is
\[
   \{  x \, |  \, x=-2 \hskip 0.2 in \text{or} \hskip 0.2 in x=8 \}.
\]
\end{explanation}
\end{question}

\begin{question} \label{Qkfkfe3}
 Let 
\[
      f(x) = 8 - 10x - x^2 
\]
and solve the equation
\[
    f(x) = -f(-1) .
\]
\begin{enumerate}

\item Write your solution as a set using set-builder notation.

\item Do not use the quadratic formula. Complete the square instead.

\item Explain the idea behind completing the square with a picture. 

\end{enumerate}

\begin{explanation}
We start by evaluating $f(-1)$:

\begin{align*}
    f(-1)  &= 8 - 10(-1) - (-1)^2  \\
             &= 8 + 10 -1 \\
             &=17.
\end{align*}

Then solving the equation $f(x) = -f(-1)$ means to solve the equation
\[
  8 - 10x - x^2  = -17,
\]
or equivalently the equation
\[
     x^2 + 10x = 25.
\]
Before completing the square, we rewrite this equation as
\[
    x^2 + 5x + 5x = 25.
\]
Now the solution continues almost exactly like the explanation to Example 1 in the chapter \href{https://ximera.osu.edu/precalc1/PreCalculus1/SolvingQuadraticEqs/SolvingQuadraticEqs}{Solving Quadratic Equations} in our class notes. I'll leave it to you to finish the solution from here.
\end{explanation}
\end{question}

 A line passes through the points $(2,1)$ and $(-1,-8)$.

    100x^2 - 300x = -90.
\]

\begin{question} \label{Qpv33f}
Let
\[
  f(x) = \frac{4x-3}{3x+2}
\]
and
\[
   g(x) = x^2  - 6x -1.
\]

Use the methods of this class to solve the equation
\[
   g(f(x)) = 6.
\]
End with a concluding sentence that summarizes the original question and states the solution set.

\begin{explanation}
To hide the composition, let $u=f(x)$. Then the equation $g(f(x))=6$ becomes
\[
     g(u) = u^2 - 6u - 1 = 6,
\]  
or equivalently
\[
   u^2 - 6u - 7 = 0.
\]
So
\[
    (u-7)(u+1)=0
\]
and $u=7$ or $u=-1$.

So $x$ is a solution to the equation $g(f(x)) = g(u) = 6$ if and only if
\[
 u = f(x) = 7 \hskip 0.3 in \text{or} \hskip 0.3 in u = f(x) = -1.
\]

I'll leave the rest for you, but the result is that the solution set to the equation $g(f(x))=6$ is
\[
 \{  x \, | \, x= -1 \text{ or } x = 1/7  \}.
\]

\end{explanation}
\end{question}

\section{Quiz 5B}

\begin{question} \label{QQA232}
\begin{enumerate}

\item Complete the following sentence.

The number $x$ is in the domain of the composition $h\circ g$ if and only if ...


\item  Let 
\[
 g(x) = \frac{2x}{x^2+8x+7}
\]
and
\[
   h(x) =      \frac{x-4}{x+1}.
\]
Use $u$-substitution and the methods of this class to find the domain of the composition $h\circ g$.

End with a concluding sentence that summarizes the original question and states domain as a set.

\end{enumerate}

\begin{explanation}
\begin{enumerate}
\item The number $x$ is in the domain of the composition $h\circ g$ if and only if
\begin{enumerate}
\item $x$ is in the domain of $g$ and

\item $g(x)$ is in the domain of $h$.
\end{enumerate}

\item We first find the domain of $g$. 

The number $x$ is in the domain of $g$ if and only if
\[
  x^2 + 8x + 7 \neq 0, 
\]
or equivalently if and only if
\[
   (x+7)(x+1) \neq 0.
\]
Since the product of two numbers is non-zero if and only if each factor is non-zero, the last inequality is equivalent to saying
\[
 x + 7 \neq 0  \hskip 0.3in \text{and} \hskip 0.3 in  x+1 \neq 0.
\]
So $x$ is in the domain of $g$ if and only if
\[
  x  \neq -7  \hskip 0.3in \text{and} \hskip 0.3 in  x \neq -1.
\]

Next we find conditions for $g(x)$ to be in the domain of $h$. 

The number $u=g(x)$ is in the domain of $h$ if and only if 
\[
  u+1 \neq 0,
\]
or equivalently if and only if
\[
    u = g(x) \neq -1.
\]

So $u=g(x)$ is in the domain of $h$ if and only if
\[
     g(x) = \frac{2x}{x^2+8x+7} \neq -1.
\]
This is equivalent to the inequality
\[
  2x \neq -x^2 -8x -7 ,
\]
or equivalently
\[
  x^2 +10x \neq -7.
\]

Completing the square gives
\[
  x^2 + 10x + 5^2 \neq -7 + 5^2,
\]
or equivalently
\[
 (x+5)^2 \neq 18.
\]

So $g(x)\neq -1$ if and only if
\[
   x \neq -5 + \sqrt{18} \hskip 0.3in \text{and} \hskip 0.3 in x \neq -5 - \sqrt{18}.
\]

Our conclusion is that the domain of the composition $h(g(x))$ is the set
\[
 \{  x \, | \, x\neq -7 \text{ and } x\neq -1 \text{ and }   x \neq -5 + \sqrt{18} \text{ and }  x \neq -5 - \sqrt{18}     \}.
\]

\end{enumerate}
\end{explanation}
\end{question}


\section{Quiz 6A}

\begin{question} \label{Q49v93f3}
\begin{enumerate}

\item[(a)] Use the methods of this class to find an expression for the inverse of the function
\[
    y = f(x) = \frac{2x-5}{x+3}.
\]

\item[(b)] Show algebraically that $f^{-1}$ undoes $f$. Do this on the back.

\item[(c)] Find the range of $f$.

\end{enumerate}

\begin{explanation}
\begin{enumerate}
\item If 
\[
 y = \frac{2x-5}{x+3} ,
\]
then
\[
 y(x+3) = 2x - 5
\]
and
\[
  yx + 3y = 2x - 5.
\]
So
\[
 2x - yx = 3y + 5
\]
and
\[
  (2-y)x = 3y + 5.
\]
So
\[
 x = \frac{3y+5}{2-y}.
\]

Our conclusion is that  for the inverse of the function
\[
    y = f(x) = \frac{2x-5}{x+3}.
\]
is the function
\[
x = f^{-1}(y) = \frac{3y+5}{2-y}.
\]

\item To show $f^{-1}$ undoes $f$, we need to show $f^{-1}(f(x)) = x$ for all $x$ in the domain of $f$ (ie. for all $x\neq 3$). Here's the algebra.

\begin{align*}
  f^{-1}(f(x))  &= f \left( \frac{2x-5}{x+3} \right) \\
                     & = \frac{3 \left( \frac{2x-5}{x+3}\right) + 5 }{2 - \left(  \frac{2x-5}{x+3} \right)} \\
                     &= \frac{3(2x-5) + 5(x+3)}{2(x+3) - (2x-5)} \text{ since } x \neq -3 \\
                     &= \frac{11x}{11}\\
                     &=x .
\end{align*}

\emph{Note:} To get from the second to the third line we multiplied the expression in the second line by  $(x+3)/(x+3) = 1$.

\item The range of the function $y=f(x)$ is the domain of the function $x=f^{-1}(y)$ and is therefore equal to the set
\[
  \{  y \, | \, y \neq 2 \}.
\]

\end{enumerate}
\end{explanation}
\end{question}

\begin{question} \label{Q4v22dd3}
\begin{enumerate}

\item[(a)] Use complete sentences to describe the action of the function
\[
    y = f(x) = 6 + \frac{5}{x-2}.
\]
on its input.

\item[(b)] Use your description in part (a) to describe the action of $f^{-1}$ on its input.

\item[(c)] Use your desription in part (b) to find an expression for the function $x = f^{-1}(y)$.

\item[(d)] Show algebraically that $f^{-1}$ undoes $f$. 

\end{enumerate}

\begin{explanation}
\begin{enumerate}
\item The function $f$ first subtracts $2$ from its input. It then divides the result into $12$ and adds $6$.

\item The function $f^{-1}$ first subtracts $6$ from its input. It then divides the result into $12$ and adds $2$.

\item So
\[
   x = f^{-1}(y) = \frac{12}{y-6} + 2.
\]

\item To show $f^{-1}$ undoes $f$, we need to show $f^{-1}(f(x)) = x$ for all $x$ in the domain of $f$ (ie. for all $x\neq 2$). Here's the algebra.

\begin{align*}
  f^{-1}(f(x))  &= f^{-1} \left(  6 + \frac{5}{x-2} \right) \\
                     &= \frac{12}{ 6 + \frac{12}{x-2} - 6} +2 \\
                     &= \frac{12}{\frac{12}{x-2}} + 2 \\
                     &= 12\left( \frac{x-2}{12} \right) +2  \text{ since } x \neq 2\\  
                     &= x-2 + 2 \\
                    &= x.
\end{align*}


\end{enumerate}
\end{explanation}
\end{question}

\section{Quiz 7A}

\begin{question} \label{Qpve31aa}
 Let 
\[
      y =  f(x) = -\sqrt{9-x^2} \, , \, -3 \leq x \leq 0.
\]

\begin{enumerate}

\item Find an expression for the function 
\[
   x = f^{-1}(y).
\]

\item Show algebraically that $f^{-1}$ undoes $f$. Do all your work on the back.
\end{enumerate}

\begin{explanation}
\begin{enumerate}
\item Since
\[
  y = -\sqrt{9-x^2},
\]
\[
  y^2 = \left( -\sqrt{9-x^2} \right)^2 ,
\]
and
\[
  y^2 = 9-x^2.
\]
So
\[
   x^2 = 9 - y^2
\]
and 
\[
   x =\pm \sqrt{9-y^2}.
\]

But since $x$ is in the domain of $f$, we know $-3\leq x \leq 0$. So 
\[
   x = - \sqrt{9-y^2}.
\]

And the inverse of the function 
\[
      y =  f(x) = -\sqrt{9-x^2} \, , \, -3 \leq x \leq 0.
\]
is the function
\[
  x = f^{-1}(y) = - \sqrt{9-y^2} \, , \, -3\leq y \leq 0.
\]
Note the domain of the function $f^{-1}$ is the range of the function $f$. Note also that the function $f$ is its own inverse.

\item To show $f^{-1}$ undoes $f$, we need to show $f^{-1}(f(x)) = x$ for all $x$ in the domain of $f$ (ie. for $-3\leq x \leq 0$).
Here's the algebra.

\begin{align*} 
  f^{-1}(f(x)) &= f^{-1}\left(  -\sqrt{9-x^2} \right)  \\
                    &= - \sqrt{9 - \left(  -\sqrt{9-x^2} \right)^2} \\
                    &= - \sqrt{9 - (9-x^2)} \\
                    &= - \sqrt{x^2} \\
                    &= - | x | .
\end{align*}
To continue simplifying, we need to go back to the domain of $f$. Since $-3\leq x \leq 0$, 
\[
  | x | = -x.
\]
So
\begin{align*}
  f^{-1}(f(x)) &= - | x |  \\
                    &= - (-x) \\
                    &= x.
\end{align*}

\end{enumerate}
\end{explanation}

\end{question}
